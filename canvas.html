Key concept — manual enter/update/exit in transition()

function transition(newData) {
  // D3's declarative join, re-implemented manually on a particle array
  const active = particles.filter(p => p.phase !== 'exiting');
  const keep   = Math.min(active.length, newData.length);
  const next   = [];

  // Keep — move existing dots to their new position
  for (let i = 0; i < keep; i++) {
    next.push({
      ...active[i],
      dx: newData[i].x, dy: newData[i].y,
      toCx: xPx(newData[i].x), toCy: yPx(newData[i].y),
      phase: 'active',
    });
  }
  // Enter — new dots grow from radius 0 at their target position
  for (let i = keep; i < newData.length; i++) {
    const cx = xPx(newData[i].x), cy = yPx(newData[i].y);
    next.push({ fromCx: cx, toCx: cx, fromCy: cy, toCy: cy, phase: 'entering' });
  }
  // Exit — removed dots shrink to radius 0, then are discarded
  for (let i = keep; i < active.length; i++) {
    next.push({ ...active[i], phase: 'exiting' });
  }

  particles = next;
  startAnim(); // kicks off a requestAnimationFrame loop
}
